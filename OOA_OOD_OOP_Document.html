3.1 หลักการ 4 ประการของ OOP ในระบบ AirCare Pro

#หลักการภาษาอังกฤษคำอธิบายตัวอย่างในระบบ

3.2 Encapsulation – การห่อหุ้มข้อมูล

🔒 Encapsulation ในคลาส Customer

ทุก attribute ถูกประกาศเป็น private ไม่สามารถเข้าถึงโดยตรงจากภายนอก ต้องผ่าน getter/setter เท่านั้น

class Customer extends User {
   // ❌ ภายนอกไม่สามารถเข้าถึงได้โดยตรง
   private string $address;
   private float $latitude;
   private float $longitude;

   // ✅ ต้องเข้าถึงผ่าน getter (อ่านอย่างเดียว)
   public function getLatitude(): float {
      return $this->latitude;
   }

   // ✅ ปรับค่าพิกัดผ่าน setter
   public function setCoordinates(float $lat, float $lng): void {
      $this->latitude = $lat;
      $this->longitude = $lng;
   }

   // ✅ Business Logic ซ่อนอยู่ใน method
   public function getGoogleMapsUrl(): string {
      return "https://www.google.com/maps?q={$this->latitude},{$this->longitude}";
   }
}

3.3 Inheritance – การสืบทอดคลาส

🧬 Inheritance – User เป็น Abstract Base Class

คลาส User เป็น Abstract Class กำหนดโครงสร้างพื้นฐาน คลาสลูก 3 คลาสสืบทอดและขยายความสามารถ

// คลาสแม่ (Abstract)
abstract class User {
   protected ?int $id;
   protected string $name;
   protected string $phone;
   protected ?string $email;
   abstract public function getRole(): string; // บังคับ implement
}

class Customer extends User {
   public function getRole(): string { return 'Customer'; }
}

class Technician extends User {
   public function getRole(): string { return 'Technician'; }
}

class Admin extends User {
   public function getRole(): string { return 'Admin'; }
}

3.4 Polymorphism – พหุสัณฐาน

🎭 Polymorphism – Method เดียว ผลลัพธ์ต่างกัน

getRole() ถูก override ในทุกคลาสลูก และ getStatusBadge() ใน ServiceOrder ทำงานต่างกันตาม state ปัจจุบัน

// Polymorphism: getRole() – ผลลัพธ์ต่างกันตามคลาส
$users = [new Customer(...), new Technician(...), new Admin(...)];
foreach ($users as $user) {
   echo $user->getRole(); // Customer / Technician / Admin
}

// Polymorphism: getStatusBadge() ใน ServiceOrder – เปลี่ยนตาม state
public function getStatusBadge(): string {
   return match ($this->status) {
      'รอดำเนินการ' => '<span class="badge-warning">🕒 รอดำเนินการ</span>',

3.5 Abstraction – การซ่อนรายละเอียด

🎯 Abstraction – Abstract Class และ Service Layer

ผู้ใช้งาน BookingManager ไม่ต้องรู้รายละเอียด SQL ภายใน เพียงเรียก method ที่มีความหมายชัดเจน

// Abstraction ระดับ 1: Abstract Class บังคับ interface
abstract class User {
   abstract public function getRole(): string;
}

// Abstraction ระดับ 2: Service Layer ซ่อน SQL
class BookingManager {
   public function createBooking(array $data): array {
      // ภายใน: สร้าง Customer → ServiceOrder → Payment → บันทึก DB
      $customer = $this->saveCustomer($data);
      $order = $this->saveOrder($customer, $data);
      $payment = $this->savePayment($order);
      return ['success' => true, 'booking_code' => $order->getBookingCode()];
   }
}

// Abstraction ระดับ 3: Static method ซ่อน logic คำนวณราคา
AirConditioner::getBasePriceForType('แอร์ติดผนัง'); // คืนค่า 500.0
AirConditioner::getBasePriceForType('แอร์สี่ทิศทาง (Cassette)'); // คืนค่า 1200.0
